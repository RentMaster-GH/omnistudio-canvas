# ==========================================
# STAGE 1: Build Rust WebAssembly Binary
# ==========================================
FROM rust:1.80-slim as wasm-builder

# Install Node.js, npm, and curl
RUN apt-get update && apt-get install -y curl nodejs npm pkg-config libssl-dev

# Install wasm-pack toolchain
RUN curl https://rustwasm.github.io/wasm-pack/installer/init.sh -sSf | sh

WORKDIR /app
COPY . .

# Compile Rust to WebAssembly
WORKDIR /app/omni_engine
RUN wasm-pack build --target web

# ==========================================
# STAGE 2: Production Lightweight Node Runner
# ==========================================
FROM node:20-alpine as runner

WORKDIR /app

# Copy root application files and compiled Wasm output from stage 1
COPY --from=wasm-builder /app /app

# Expose server port
EXPOSE 3000
ENV NODE_ENV=production
ENV PORT=3000

# Start OmniStudio Collaboration & Rendering Server
CMD ["npm", "start"]
